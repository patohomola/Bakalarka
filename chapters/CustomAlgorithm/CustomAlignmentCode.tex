from collections import Counter
from typing import Optional


class SeqDnaMap:
    database: dict[int, str]
    dna_maxSize: int = -1

    def __init__(self) -> None:
        self.database = {}

    def add_dna(self, dna: str, dna_index: int) -> None:
        self.database[dna_index] = dna

    def Get_dna(self, dna_index: int) -> str:
        return self.database[dna_index]

    def SetMaxIndex(self, maxIndex: int) -> None:
        self.dna_maxSize = maxIndex

    def GetMaxIndex(self) -> int:
        return self.dna_maxSize

class Line:
    database: SeqDnaMap
    prev_Dna_line: Optional[Line]
    dna_index: int = 0
    dna_back_track_seq: str = ""
    seq_index: int = 0
    line_score: int = 0

    def __init__(self, seqDnaRef: SeqDnaMap, dna_index: int) -> None:
        self.database = seqDnaRef
        self.dna_index = dna_index
        self.dna_back_track_seq = ""
        self.seq_index = 0
        self.line_score = 0
        self.dna_maxIndex = -1
        self.prev_Dna_line = None

    def getNextNucleoid(self) -> Optional[str]:
        """Return next nucleotide in dna_seq, or None if at end."""
        dna_seq = self.database.Get_dna(self.dna_index)
        '''if self.database.GetMaxIndex() != -1:
            if self.seq_index >= self.database.GetMaxIndex():
                return None'''
        if self.seq_index >= len(dna_seq):
            return None
        return dna_seq[self.seq_index]

    def linesFromNextNucleod(self, dnaNucleoid: Optional[str]) -> list[Line]:
        """
        Compare the next nucleotide with dnaNucleoid.
        Return possible continuation lines (with errors).
        """
        myNucleoid = self.getNextNucleoid()

        # --- PERFECT MATCH ---
        if dnaNucleoid == myNucleoid:
            #new_line = Line(self.database, self.dna_index)
            self.seq_index +=1
            self.dna_back_track_seq += dnaNucleoid
            self.line_score += 1
            return [self]

        # --- ERRORS ---
        errorLines = []

        # BASELINE ERROR (replace)
        replace_line = Line(self.database, self.dna_index)
        replace_line.seq_index = self.seq_index + 1
        replace_line.prev_Dna_line = self
        replace_line.line_score = self.line_score - 4
        replace_line.dna_back_track_seq = ""+ myNucleoid #myNucleoid.lower() + "_"
        errorLines.append(replace_line)

        # --- DELETION ---
        dna_del = Line(self.database, self.dna_index)
        dna_del.seq_index = self.seq_index       # skip the match (deletion)
        dna_del.prev_Dna_line=self
        dna_del.line_score = self.line_score - 4
        dna_del.dna_back_track_seq = "" + "_"
        errorLines.append(dna_del)

        # --- INSERTION ---
        dna_ins = Line(self.database, self.dna_index)
        dna_ins.prev_Dna_line=self
        dna_ins.seq_index = self.seq_index
        dna_ins.line_score = self.line_score
        dna_ins.dna_back_track_seq = ""

        # add inserted bases until match
        while True:
            next_base = dna_ins.getNextNucleoid()
            if next_base is None:
                break  # reached end of sequence

            if next_base == dnaNucleoid:
                # match finally found
                dna_ins.dna_back_track_seq += next_base
                dna_ins.seq_index += 1
                dna_ins.line_score += 1
                break

            # still inserting errors
            dna_ins.dna_back_track_seq += next_base.lower()
            dna_ins.seq_index += 1
            dna_ins.line_score -= 4

        errorLines.append(dna_ins)
        return errorLines

    def getBacktrackRecursive(self) -> str:
        if self.prev_Dna_line is None:
            return self.dna_back_track_seq
        return self.prev_Dna_line.getBacktrackRecursive() + self.dna_back_track_seq

    def dnaLenSeq(self) -> int:
        return len(self.database.Get_dna(self.dna_index))

    def getUnalignedPart(self) -> str:
        dna_seq = self.database.Get_dna(self.dna_index)
        return dna_seq[self.seq_index:]

def deduplicate_lines(lines: list[Line]) -> list[Line]:
    """Deduplicate by (dna_index, seq_index), keep highest score."""
    dedup: dict[tuple[int, int], Line] = {}
    for l in lines:
        key = (l.dna_index, l.seq_index)
        if key not in dedup or l.line_score > dedup[key].line_score:
            dedup[key] = l
    return list(dedup.values())

def Select_Best_Align(lines: list[Line]) -> list[Line]:
    """Deduplicate by (dna_index, seq_index), keep highest score."""
    dedup: dict[int, Line] = {}
    for l in lines:
        key = l.dna_index
        if key not in dedup or l.line_score > dedup[key].line_score:
            dedup[key] = l
    return list(dedup.values())

def NaiveMultiAlignment(lines: list[Line], max_score_drop: int = 20) -> list[Line]:
    """
    Naive multi-sequence alignment (beam search style).

    lines = list[Line]
    """
    finished: list[Line] = []
    max_distance = 0
    step = 0
    consensus = ""
    while True:
        step += 1

        # --- Beam pruning: drop lines far behind best ---
        best_per_index: dict[int, int] = {}
        best_Consensus: dict[int, Optional[str]] = {}


        for l in lines:
            if l.dna_index not in best_per_index:
                best_per_index[l.dna_index] = l.line_score
                best_Consensus[l.dna_index] = l.getNextNucleoid()
            else:
                best_per_index[l.dna_index] = max(best_per_index[l.dna_index], l.line_score)

        for l in finished:
            if l.dna_index not in best_per_index:
                best_per_index[l.dna_index] = l.line_score
            else:
                best_per_index[l.dna_index] = max(best_per_index[l.dna_index], l.line_score)

        # Drop only lines that fall too far behind *their own dna_index group*
        filtered: list[Line] = []
        for l in lines:
            if l.line_score >= best_per_index[l.dna_index] - max_score_drop:
                filtered.append(l)

        lines = filtered

        for line in lines:
            if line.seq_index >= line.dnaLenSeq():
                finished.append(line)
            if max_distance > 0:
                finished.append(line)

        finished = deduplicate_lines(finished)

        # Filter only active (not finished) lines
        active = [l for l in lines if l.seq_index < l.dnaLenSeq()]

        print("step ", step)
        print("size", len(active))


        if len(active) == 0:
            break

        # --- Majority vote nucleoid (most common next nucleotide) ---
        #next_bases = [l.getNextNucleoid() for l in active]
        base_counts = Counter(best_Consensus.values())
        #weighted_counts = Counter()
        #for l in active:
        #    base = l.getNextNucleoid()
        #    if base:
        #        weighted_counts[base] += max(1,l.line_score)

        #most_common = weighted_counts.most_common(1)[0][0]
        #base_counts = weighted_counts
        most_common = base_counts.most_common(1)[0][0]
        consensus+=most_common

        most_common_base, most_common_count = base_counts.most_common(1)[0]
        fraction_on_most_common = most_common_count / len(best_Consensus.values())
        print(f"{fraction_on_most_common:.4f}")

        if (fraction_on_most_common < 0.5 and step>5):
            # Its clearly there are no same lines at all
            max_distance = step
            if lines:
                lines[0].database.SetMaxIndex(step)
            for line in lines:
                finished.append(line)
            finished = deduplicate_lines(finished)
            break

        #print(best_Consensus.values())
        # --- Expand all lines using this nucleotide ---
        new_lines: list[Line] = []
        for l in active:
            extended = l.linesFromNextNucleod(most_common)
            new_lines.extend(extended)


        lines = deduplicate_lines(new_lines)



        # Safety: If no lines left, stop
        if not lines:
            break

    # Return all final results sorted by best score
    finished=Select_Best_Align(finished)
    finished.sort(key=lambda l: l.line_score, reverse=True)

    print("consensus =", consensus)
    return finished

def align_backtracks(lines: list[str]) -> list[str]:
    seqs: list[list[str]] = [list(s) for s in lines]
    pos = [0] * len(seqs)
    out = [""] * len(seqs)
    n = len(seqs)

    while any(pos[i] < len(seqs[i]) for i in range(n)):
        col = [seqs[i][pos[i]] if pos[i] < len(seqs[i]) else None for i in range(n)]
        has_lower = any(c and c.islower() for c in col)

        for i, c in enumerate(col):
            if has_lower:
                if c and c.islower():
                    out[i] += c
                    pos[i] += 1
                else:
                    out[i] += "_"
            else:
                out[i] += c if c else "_"
                pos[i] += 1

    return out


def DataBeamAlgorithm(reads: list[str], prooning: int = 40) -> list[str]:
    dna_database = SeqDnaMap()
    for i, dna in enumerate(reads):
        dna_database.add_dna(dna, i)
    lines = [Line(dna_database, i) for i, dna in enumerate(reads)]
    Out = NaiveMultiAlignment(lines, prooning)
    Align = align_backtracks([res.getBacktrackRecursive() for res in Out])
    return Align
