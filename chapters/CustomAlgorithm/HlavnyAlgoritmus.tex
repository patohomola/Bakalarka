        best_per_index: dict[int, int] = {}

Multiple sequence alignment (MSA) je problém zarovnania viacerých sekvencií, ktorý sa často vyskytuje v bioinformatike. Cieľom MSA je nájsť optimálne zarovnanie medzi viacerými sekvenciami, ktoré minimalizuje počet nesúladov a medzier. V tejto časti popisujem vlastný algoritmus na zarovnanie viacerých sekvencií, ktorý som implementoval v Pythone.
\\
\\Pre tento Algoritmus predpokladám, že máme k dispozícii sequencie DNA z zhlukovania od algoritmu clover, ktory zhlukuje sekvencie, podla toho, ako začínajú. Takze mozeme predpokladat, ze zaciatok sekvencii v jednom zhluku je rovnaky a zarovnanie zacina od zaciatku a dalej sa rozvetvuje kvoli chybam, ktore sa vyskytuju pri citani dna.
Algoritmus využíva prístup podobný beam search, kde pre každú sekvenciu udržiavame množinu možných zarovnaní (vetiev) a iteratívne ich rozširujeme o ďalší znak z konsenzuálnej sekvencie. Pri každom kroku vyberáme najčastejší znak (konsenzus) a generujeme nové vetvy pre každý možný typ chyby (zhoda, zámienka, vloženie, vynechanie). Vetvy sú následne filtrované na základe skóre, aby sme udržali len tie najpravdepodobnejšie zarovnania. Tento proces pokračuje, kým nedosiahneme koniec všetkých sekvencií alebo kým sa seqvencie nezacnú výrazne odlišovat od seba, čo indikuje, že neexistuje jasný konsenzus.

\begin{algorithm}
\caption{Stav jednej vetvy zarovnania}\label{alg:alignment-state}
\begin{algorithmic}
    \State \textbf{record} $AlignmentState$
    \State \, $score$ : aktuálne skóre vetvy
    \State \, $dna$ : spracovávaná DNA sekvencia
    \State \, $position$ : aktuálna pozícia v sekvencii
    \State \, $backtrack$ : postupnosť symbolov $\{=, I, D, M\}$

    \Function{NextBase}{$state$}
        \If{$state.position < |state.dna|$}
            \State \Return $state.dna[state.position]$
        \EndIf
        \State \Return $None$
    \EndFunction

    \Function{UpdateState}{$state, \Delta score, advance, symbol$}
        \State $state.score \gets state.score + \Delta score$
        \If{$advance$}
            \State $state.position \gets state.position + 1$
        \EndIf
        \State append $symbol$ to $state.backtrack$
    \EndFunction
\end{algorithmic}
\end{algorithm}

              

\begin{algorithm}
\caption{Generovanie kandidátov pre jednu vetvu}\label{alg:generate-candidates}
\begin{algorithmic}
    \Function{GenerateCandidates}{$consensus, state$}
        \If{\Call{NextBase}{$state$} $= consensus$}
            \State \Call{UpdateState}{$state$, $MatchReward$, true, $=$}
            \State \Return $[state]$
        \EndIf

        \State $mismatch \gets$ kópia $state$
        \State \Call{UpdateState}{$mismatch$, $-MismatchPenalty$, true, $M$}

        \State $deletion \gets$ kópia $state$
        \State \Call{UpdateState}{$deletion$, $-DeletionGapPenalty$, false, $D$}

        \State $insertion \gets$ kópia $state$
        \While{\Call{NextBase}{$insertion$} $\neq consensus$}
            \State \Call{UpdateState}{$insertion$, $-InsertionGapPenalty$, true, $I$}
        \EndWhile
        
        \State \Call{UpdateState}{$insertion$, $MatchReward$, true, $=$}
        

        \State \Return $[mismatch, insertion, deletion]$
    \EndFunction
\end{algorithmic}
\end{algorithm}    

\begin{algorithm}
\caption{Multi-Alignment}\label{alg:cap}
\begin{algorithmic}
    \Require zoznam stavov $lines$
    \While{true}
        \State $bestCandidates \gets$ najlepšia vetva pre každú vstupnú sekvenciu v $lines$
        \State $consensus \gets$ najčastejšia hodnota \Call{NextBase}{$candidate$} pre $candidate \in bestCandidates$

        \State $newLines \gets [\ ]$
        \ForAll{$line \in lines$}
            \State pridaj všetky prvky z \Call{GenerateCandidates}{$consensus, line$} do $newLines$
        \EndFor
        \State $lines \gets newLines$
        \State odstráň vetvy, ktoré pre rovnakú sekvenciu a rovnakú pozíciu maju horšie skóre než iná vetva
        \State odstráň vetvy so skóre nižším než $threshold\_score$

        \If{menej než $threshold\_ratio$ z $bestCandidates$ podporuje $consensus$}
            \State \Return $bestCandidates$
        \EndIf
    \EndWhile
\end{algorithmic}
\end{algorithm}
